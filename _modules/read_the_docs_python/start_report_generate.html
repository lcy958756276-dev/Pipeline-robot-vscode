#!/usr/bin/env python
"""
start_report_generate.py
========================

This module provides a ROS node to start or stop the report generation process
using a ROS launch file (`auto_report.launch`). It listens to the 
`whether_report_generate` topic and starts or terminates the process accordingly.

Modules:
--------
- rospy: ROS Python client library
- std_msgs.msg: Standard ROS message types
- subprocess: To launch external processes
"""

import rospy
from std_msgs.msg import Bool
import subprocess

roslaunch_process = None

def handle_roslaunch(is_start: bool) -> None:
    """
    Start or stop the report generation process.

    Parameters
    ----------
    is_start : bool
        True to start the process, False to terminate it.

    Notes
    -----
    Uses a global variable `roslaunch_process` to track the subprocess.
    """
    global roslaunch_process
    if is_start:
        if roslaunch_process is None:
            try:
                rospy.loginfo("===================================subprocess start========================")
                roslaunch_process = subprocess.Popen(['roslaunch', 'pipeline_robot', 'auto_report.launch'])
                rospy.loginfo("report_generate started.")
            except subprocess.CalledProcessError as e:
                rospy.logerr("Failed to start report_generate: %s", e)
        else:
            rospy.loginfo("report_generate is already running.")
    else:
        if roslaunch_process is not None:
            roslaunch_process.terminate()
            roslaunch_process = None
            rospy.loginfo("report_generate terminated.")
        else:
            rospy.loginfo("No report_generate process to terminate.")

def report_generate_callback(msg: Bool) -> None:
    """
    Callback function for the ROS subscriber.

    Parameters
    ----------
    msg : std_msgs.msg.Bool
        Message received from the `whether_report_generate` topic.
    """
    rospy.loginfo("Received report_generate: %s", msg.data)
    handle_roslaunch(msg.data)

def listener() -> None:
    """
    Initializes the ROS node and subscribes to the topic.

    Listens to `whether_report_generate` topic and calls 
    `report_generate_callback` when a message is received.
    """
    rospy.init_node('report_generate_listener', anonymous=True)
    rospy.Subscriber("whether_report_generate", Bool, report_generate_callback)
    rospy.spin()



if __name__ == '__main__':
    listener()
