101void publishOdometry(ros::Publisher& odom_pub, tf::TransformBroadcaster& tf_broadcaster)
102{
103 ros::Time current_time = ros::Time::now();
104 double dt = (current_time - last_time).toSec();
105
106 // 处理里程计重置
107 if (reset_odom)
108 {
109 x = 0.0;
110 y = 0.0;
111 theta = 0.0;
112 reset_odom = false; // 重置完成
113 }
114
115 double d_distance = distance; // 计算行进距离
116 x += d_distance; // 更新 x 坐标
117 // y += d_distance * sin(theta); // 更新 y 坐标（未使用）
118 theta = 0; // 更新角度
119
120 // 创建里程计消息
121 nav_msgs::Odometry odom;
122 odom.header.frame_id = "odom";
123 odom.pose.pose.position.x = x;
124 odom.pose.pose.position.y = 0;
125 odom.pose.pose.position.z = 0.0;
126 odom.pose.pose.orientation = tf::createQuaternionMsgFromYaw(theta);
127 odom.child_frame_id = "base_link";
128 odom.twist.twist.linear.x = d_distance / dt;
129
130 // 发布里程计
131 odom_pub.publish(odom);
132
133 // 广播 tf 变换
134 tf::Transform transform;
135 transform.setOrigin(tf::Vector3(x, y, 0.0));
136 tf::Quaternion q;
137 q.setRPY(0, 0, theta);
138 transform.setRotation(q);
139 tf_broadcaster.sendTransform(tf::StampedTransform(transform, ros::Time::now(), "odom", "base_link"));
140
141 // 重置行进距离
142 distance = 0.0;
143 last_time = current_time;
144}
