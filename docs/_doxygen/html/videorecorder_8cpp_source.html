208void recordingThread()
209{
210 AVFrame *yuvFrame = av_frame_alloc();
211 yuvFrame->format = AV_PIX_FMT_YUV420P;
212 yuvFrame->width = m_width;
213 yuvFrame->height = m_height;
214 if (av_frame_get_buffer(yuvFrame, 32) < 0) {
215 qWarning() << "Failed to alloc yuv buffer";
216 av_frame_free(&yuvFrame);
217 return;
218 }
219
220 AVPacket *packet = av_packet_alloc();
221 if (!packet) {
222 av_frame_free(&yuvFrame);
223 return;
224 }
225
226 while (!m_stopRequested) {
227 QImage frame;
228
229 {
230 QMutexLocker locker(&m_mutex);
231 if (m_frameQueue.isEmpty()) {
232 m_condition.wait(&m_mutex, 200); // 队列为空则等待
233 continue;
234 }
235 frame = m_frameQueue.takeFirst();
236 }
237
238 if (frame.isNull()) continue;
239
240 uint8_t *srcData[4] = { const_cast<uint8_t*>(frame.bits()), nullptr, nullptr, nullptr };
241 int srcLinesize[4] = { frame.bytesPerLine(), 0, 0, 0 };
242
243 sws_scale(m_swsContext, srcData, srcLinesize, 0, m_height,
244 yuvFrame->data, yuvFrame->linesize);
245
246 yuvFrame->pts = m_frameCount++;
247
248 if (avcodec_send_frame(m_codecContext, yuvFrame) < 0) {
249 qDebug() << "Error sending frame to encoder";
250 continue;
251 }
252
253 while (avcodec_receive_packet(m_codecContext, packet) == 0) {
254 av_packet_rescale_ts(packet, m_codecContext->time_base, m_videoStream->time_base);
255 packet->stream_index = m_videoStream->index;
256 av_interleaved_write_frame(m_formatContext, packet);
257 av_packet_unref(packet);
258 }
259 }
260
261 // flush encoder
262 avcodec_send_frame(m_codecContext, nullptr);
263 while (avcodec_receive_packet(m_codecContext, packet) == 0) {
264 av_packet_rescale_ts(packet, m_codecContext->time_base, m_videoStream->time_base);
265 packet->stream_index = m_videoStream->index;
266 av_interleaved_write_frame(m_formatContext, packet);
267 av_packet_unref(packet);
268 }
269
270 av_write_trailer(m_formatContext);

272 av_frame_free(&yuvFrame);
273 av_packet_free(&packet);
