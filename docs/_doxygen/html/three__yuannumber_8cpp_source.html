133bool connectToServer(const std::string& ip, int port) {
134 std::lock_guard<std::mutex> lock(g_conn_mutex);
135 ROS_INFO_STREAM("Connecting to " << ip << ":" << port);
136
137 g_sockfd = socket(AF_INET, SOCK_STREAM, 0);
138 if (g_sockfd < 0) {
139 ROS_ERROR_STREAM("Socket error: " << strerror(errno));
140 return false;
141 }
142
143 sockaddr_in serv_addr;
144 serv_addr.sin_family = AF_INET;
145 serv_addr.sin_port = htons(port);
146 if (inet_pton(AF_INET, ip.c_str(), &serv_addr.sin_addr) <= 0) {
147 ROS_ERROR("Invalid IP format");
148 return false;
149 }
150
151 if (connect(g_sockfd, (struct sockaddr*)&serv_addr, sizeof(serv_addr)) < 0) {
152 ROS_ERROR_STREAM("Connection failed: " << strerror(errno));
153 return false;
154 }
155
156 ROS_INFO("Connected to server");
157 publishStatus("CONNECTED");
158 return true;
159}

168void tcpDataLoop(ros::Publisher& raw_pub,
169 ros::Publisher& roll_pub,
170 ros::Publisher& pitch_pub,
171 ros::Publisher& heading_pub) {
172 char buffer[4096];
173 std::string remaining_data;
174
175 while (ros::ok()) {
176 ssize_t valread = read(g_sockfd, buffer, sizeof(buffer) - 1);
177 if (valread <= 0) {
178 ROS_WARN_STREAM("Read error: " << (valread == 0 ? "EOF" : strerror(errno)));
179 break;
180 }
181
182 buffer[valread] = '\0';
183 std::string raw_data = remaining_data + buffer;
184 size_t last_newline = raw_data.find_last_of('\n');
185
186 if (last_newline != std::string::npos) {
187 // 发布原始数据
188 std_msgs::String raw_msg;
189 raw_msg.data = "[RAW] " + raw_data.substr(0, 100); // 截断前100字符
190 raw_pub.publish(raw_msg);
191
192 // 解析RPH数据
193 parseAndPublishData(raw_data.substr(0, last_newline), roll_pub, pitch_pub, heading_pub);
194 remaining_data = raw_data.substr(last_newline + 1);
195 } else {
196 remaining_data = raw_data;
197 ROS_DEBUG("Buffering incomplete packet");
198 }
199 memset(buffer, 0, sizeof(buffer));
200 }
201}

203void parseAndPublishData(const std::string& data,
204 ros::Publisher& roll_pub,
205 ros::Publisher& pitch_pub,
206 ros::Publisher& heading_pub) {
207 std::istringstream iss(data);
208 std::string line;
209 float roll = 0, pitch = 0, heading = 0;
210 bool has_rph = false;
211
212 while (std::getline(iss, line)) {
213 if (line.empty()) continue;
214 ROS_DEBUG_STREAM("Processing: " << line);
215
216 try {
217 if (line.find("R=") == 0) {
218 roll = std::stof(line.substr(2));
219 has_rph = true;
220 }
221 else if (line.find("P=") == 0) {
222 pitch = std::stof(line.substr(2));
223 has_rph = true;
224 }
225 else if (line.find("H=") == 0) {
226 heading = std::stof(line.substr(2));
227 if (has_rph) {
228 std_msgs::Float32 roll_msg, pitch_msg, heading_msg;
229 roll_msg.data = roll;
230 pitch_msg.data = pitch;
231 heading_msg.data = heading;
232
233 roll_pub.publish(roll_msg);
234 pitch_pub.publish(pitch_msg);
235 heading_pub.publish(heading_msg);
236
237 ROS_DEBUG_STREAM("Published RPH: R=" << roll
238 << ", P=" << pitch
239 << ", H=" << heading);
240 has_rph = false;
241 }
242 }
243 } catch (const std::exception& e) {
244 ROS_ERROR_STREAM("Parse error: " << e.what() << " in line: " << line);
245 }
246 }
