142void imageCallback(const sensor_msgs::ImageConstPtr& msg)
143{
144 cv_bridge::CvImagePtr cv_ptr;
145 try {
146 cv_ptr = cv_bridge::toCvCopy(msg, sensor_msgs::image_encodings::BGR8);
147 } catch (cv_bridge::Exception& e) {
148 ROS_ERROR("cv_bridge exception: %s", e.what());
149 return;
150 }
151
152 cv::Mat mat = cv_ptr->image;
153 if (mat.empty()) return;
154
155 cv::Mat mat_rgb;
156 cv::cvtColor(mat, mat_rgb, cv::COLOR_BGR2RGB);
157
158 QImage img(mat_rgb.data, mat_rgb.cols, mat_rgb.rows, static_cast<int>(mat_rgb.step), QImage::Format_RGB888);
159 emit imageReceived(img.copy());
160}

167QImage Mat2QImage(cv::Mat const& src)
168{
169 QImage dest(src.cols, src.rows, QImage::Format_ARGB32);
170 const float scale = 255.0;
171
172 if (src.depth() == CV_8U) {
173 if (src.channels() == 1) {
174 for (int i = 0; i < src.rows; ++i)
175 for (int j = 0; j < src.cols; ++j)
176 dest.setPixel(j, i, qRgb(src.at<quint8>(i, j), src.at<quint8>(i, j), src.at<quint8>(i, j)));
177 } else if (src.channels() == 3) {
178 for (int i = 0; i < src.rows; ++i)
179 for (int j = 0; j < src.cols; ++j) {
180 cv::Vec3b bgr = src.at<cv::Vec3b>(i, j);
181 dest.setPixel(j, i, qRgb(bgr[2], bgr[1], bgr[0]));
182 }
183 }
184 } else if (src.depth() == CV_32F) {
185 if (src.channels() == 1) {
186 for (int i = 0; i < src.rows; ++i)
187 for (int j = 0; j < src.cols; ++j) {
188 int level = scale * src.at<float>(i, j);
189 dest.setPixel(j, i, qRgb(level, level, level));
190 }
191 } else if (src.channels() == 3) {
192 for (int i = 0; i < src.rows; ++i)
193 for (int j = 0; j < src.cols; ++j) {
194 cv::Vec3f bgr = scale * src.at<cv::Vec3f>(i, j);
195 dest.setPixel(j, i, qRgb(bgr[2], bgr[1], bgr[0]));
196 }
197 }
198 }
199 return dest;
200}
