87void tcpLoop(const std::string& ip, int port) {
88 int sockfd = socket(AF_INET, SOCK_STREAM, 0);
89 if (sockfd < 0) {
90 ROS_ERROR("Socket creation failed");
91 return;
92 }
93
94 sockaddr_in serv_addr;
95 serv_addr.sin_family = AF_INET;
96 serv_addr.sin_port = htons(port);
97 inet_pton(AF_INET, ip.c_str(), &serv_addr.sin_addr);
98
99 if (connect(sockfd, (struct sockaddr*)&serv_addr, sizeof(serv_addr)) < 0) {
100 ROS_ERROR("Connection failed to %s:%d", ip.c_str(), port);
101 return;
102 }
103
104 ROS_INFO("Connected to server %s:%d", ip.c_str(), port);
105
106 char buffer[1024];
107 std::string leftover;
108
109 while (ros::ok()) {
110 ssize_t n = read(sockfd, buffer, sizeof(buffer) - 1);
111 if (n <= 0) {
112 ROS_WARN("Server disconnected");
113 break;
114 }
115 buffer[n] = '\0';
116 std::string data = leftover + buffer;
117
118 std::istringstream iss(data);
119 std::string line;
120 while (std::getline(iss, line)) {
121 if (!line.empty()) parseAndPublish(line);
122 }
123
124 // 处理不完整的一行
125 if (data.back() != '\n')
126 leftover = data.substr(data.find_last_of('\n') + 1);
127 else
128 leftover.clear();
129
130 ros::spinOnce();
131 }
132
133 close(sockfd);
134}
