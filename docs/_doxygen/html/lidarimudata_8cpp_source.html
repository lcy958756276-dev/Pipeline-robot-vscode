118bool connectToServer(const std::string& ip, int port) {
119 std::lock_guard<std::mutex> lock(g_conn_mutex);
120 ROS_INFO_STREAM("Connecting to " << ip << ":" << port);
121
122 g_sockfd = socket(AF_INET, SOCK_STREAM, 0);
123 if (g_sockfd < 0) {
124 ROS_ERROR_STREAM("Socket error: " << strerror(errno));
125 return false;
126 }
127
128 sockaddr_in serv_addr;
129 serv_addr.sin_family = AF_INET;
130 serv_addr.sin_port = htons(port);
131 if (inet_pton(AF_INET, ip.c_str(), &serv_addr.sin_addr) <= 0) {
132 ROS_ERROR("Invalid IP format");
133 return false;
134 }
135
136 if (connect(g_sockfd, (struct sockaddr*)&serv_addr, sizeof(serv_addr)) < 0) {
137 ROS_ERROR_STREAM("Connection failed: " << strerror(errno));
138 return false;
139 }
140
141 ROS_INFO("Connected to server");
142 publishStatus("CONNECTED");
143 return true;
144}

165void tcpDataLoop(ros::Publisher& raw_pub,
166 ros::Publisher& point_pub,
167 ros::Publisher& da_pub) {
168 char buffer[4096];
169 std::string remaining_data;
170
171 while (ros::ok()) {
172 ssize_t valread = read(g_sockfd, buffer, sizeof(buffer) - 1);
173 if (valread <= 0) {
174 ROS_WARN_STREAM("Read error: " << (valread == 0 ? "EOF" : strerror(errno)));
175 break;
176 }
177
178 buffer[valread] = '\0';
179 std::string raw_data = remaining_data + buffer;
180 size_t last_newline = raw_data.find_last_of('\n');
181
182 if (last_newline != std::string::npos) {
183 // 发布原始数据
184 std_msgs::String raw_msg;
185 raw_msg.data = "[RAW] " + raw_data.substr(0, 100); // 截断前100字符
186 raw_pub.publish(raw_msg);
187
188 // 解析有效数据包
189 parseAndPublishData(raw_data.substr(0, last_newline), point_pub, da_pub);
190 remaining_data = raw_data.substr(last_newline + 1);
191 } else {
192 remaining_data = raw_data;
193 ROS_DEBUG("Buffering incomplete packet");
194 }
195 memset(buffer, 0, sizeof(buffer));
196 }
197}

206void parseAndPublishData(const std::string& data,
207 ros::Publisher& point_pub,
208 ros::Publisher& da_pub) {
209 std::istringstream iss(data);
210 std::string line;
211 geometry_msgs::PointStamped point;
212 point.header.frame_id = "sensor_frame";
213 bool has_coordinates = false;
214
215 pipeline_robot::DistanceAngle da_msg;
216 da_msg.header.stamp = ros::Time::now();
217
218 while (std::getline(iss, line)) {
219 if (line.empty()) continue;
220 ROS_DEBUG_STREAM("Processing: " << line);
221
222 try {
223 // 解析XYZ坐标
224 if (line.find("X=") == 0) {
225 point.point.x = std::stof(line.substr(2));
226 has_coordinates = true;
227 }
228 else if (line.find("Y=") == 0) {
229 point.point.y = std::stof(line.substr(2));
230 has_coordinates = true;
231 }
232 else if (line.find("Z=") == 0) {
233 point.point.z = std::stof(line.substr(2));
234 if (has_coordinates) {
235 point.header.stamp = ros::Time::now();
236 point_pub.publish(point);
237 ROS_DEBUG_STREAM("Published point: " <<
238 point.point.x << ", " <<
239 point.point.y << ", " <<
240 point.point.z);
241 has_coordinates = false;
242 }
243 }
244 // 解析Distance和Angle
245 else if (line.find("D=") == 0 && line.find("A=") != std::string::npos) {
246 size_t a_pos = line.find("A=");
247 da_msg.distance = std::stof(line.substr(2, a_pos - 2));
248 da_msg.angle = std::stof(line.substr(a_pos + 2));
249 da_msg.header.stamp = ros::Time::now();
250 da_pub.publish(da_msg);
251 ROS_DEBUG_STREAM("Published DA: D=" << da_msg.distance << ", A=" << da_msg.angle);
252 }
253 } catch (const std::exception& e) {
254 ROS_ERROR_STREAM("Parse error: " << e.what() << " in line: " << line);
255 }
256 }
257}
