137void DetectionAnalyzer::detectionCallback(const pipeline_robot::detection::ConstPtr &msg)

150void DetectionAnalyzer::publishMidGoal()
151{
152 if (!detections.empty())
153 {
154 auto min_it = std::min_element(detections.begin(), detections.end());
155 float min_detection_x = *min_it;
156
157 if (!isSimilarPublished(min_detection_x))
158 {
159 geometry_msgs::PoseStamped midgoal;
160 midgoal.header.stamp = ros::Time::now();
161 midgoal.header.frame_id = "odom";
162 midgoal.pose.position.x = min_detection_x - 0.5;
163 midgoal.pose.orientation.w = 1.0;
164 midgoal_pub.publish(midgoal);
165
166 published_positions.push_back(min_detection_x);
167 }
168 detections.erase(min_it);
169 }
170}
