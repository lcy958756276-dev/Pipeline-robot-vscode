59 void daCallback(const pipeline_robot::DistanceAngle::ConstPtr& msg) {
60 if (!odom_received_) {
61 ROS_WARN_THROTTLE(5, "Waiting for /cable/odom ...");
62 return;
63 }
64
65 ros::Time current_time = ros::Time::now();
66 float angle_rad = msg->angle * M_PI / 180.0;
67
68 pcl::PointXYZ point;
69 point.x = current_pose_.position.x;
70 point.y = msg->distance * sin(angle_rad) * 0.01;
71 point.z = msg->distance * cos(angle_rad) * (-0.01);
72
73 cloud_->points.push_back(point);
74
75 // 更新八叉树
