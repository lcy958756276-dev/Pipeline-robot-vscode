 57/**
 58 * @brief 发送 Modbus 指令帧
 59 * @param frame 指令帧数据
 60 * @param length 指令长度
 61 */
 62void Modbus_Send(uint8_t *frame, uint16_t length)
 63{
 64    if(serialPort.isOpen() && serialPort.isWritable()){
 65        for (int i = 0; i < length; i++) {
 66            char byte = static_cast<char>(frame[i]);
 67            serialPort.write(&byte, 1);
 68            serialPort.flush();
 69        }
 70    } else {
 71        qDebug() << "Modbus 发送错误";
 72    }
 73}

 75/**
 76 * @brief 计算 Modbus CRC16 校验
 77 * @param data 数据缓冲区
 78 * @param length 数据长度
 79 * @return uint16_t CRC16 校验值
 80 */
 81uint16_t Modbus_CRC16(uint8_t *data, uint16_t length)
 82{
 83    uint16_t crc = 0xFFFF;
 84    for (int pos = 0; pos < length; pos++) {
 85        crc ^= static_cast<uint16_t>(data[pos]);
 86        for (int i = 0; i < 8; i++) {
 87            if ((crc & 0x0001) != 0) {
 88                crc >>= 1;
 89                crc ^= 0xA001;
 90            } else {
 91                crc >>= 1;
 92            }
 93        }
 94    }
 95    return crc;
 96}

 50 * 持续解码 RTSP 流，转换为 QImage 并发送 frameDecoded 信号。
 51 * 断开连接时触发 connectionLost 信号，可实现自动重连。
 52 */
 53void run() {
 54    while(!isInterruptionRequested()) {
 55        if(!initFFmpeg()) {
 56            emit errorOccurred("初始化失败");
 57            break;
 58        }
 59
 60        AVFrame *frame = av_frame_alloc();
 61        bool connection_ok = true;
 62
 63        while(connection_ok && !isInterruptionRequested()) {
 64            AVPacket pkt;
 65            int ret = av_read_frame(fmt_ctx_, &pkt);
 66            if(ret < 0) {
 67                if(ret == AVERROR_EOF || avio_feof(fmt_ctx_->pb)) {
 68                    qDebug() << "RTSP连接中断";
 69                    emit connectionLost();
 70                    connection_ok = false;
 71                }
 72                break;
 73            }
 74
 75            if(pkt.stream_index == video_stream_index_) {
 76                if(avcodec_send_packet(codec_ctx_, &pkt) < 0) {
 77                    av_packet_unref(&pkt);
 78                    continue;
 79                }
 80
 81                AVFrame *rgb_frame = av_frame_alloc();
 82                uint8_t *buffer = nullptr;
 83                int num_bytes = av_image_get_buffer_size(AV_PIX_FMT_RGB32,
 84                                                         codec_ctx_->width,
 85                                                         codec_ctx_->height, 1);
 86                buffer = (uint8_t *)av_malloc(num_bytes * sizeof(uint8_t));
 87                av_image_fill_arrays(rgb_frame->data, rgb_frame->linesize, buffer,
 88                                     AV_PIX_FMT_RGB32, codec_ctx_->width, codec_ctx_->height, 1);
 89
 90                while(avcodec_receive_frame(codec_ctx_, frame) == 0) {
 91                    sws_scale(sws_ctx_,
 92                              (uint8_t const * const *)frame->data,
 93                              frame->linesize, 0, codec_ctx_->height,
 94                              rgb_frame->data, rgb_frame->linesize);
 95
 96                    QImage img(rgb_frame->data[0],
 97                               codec_ctx_->width,
 98                               codec_ctx_->height,
 99                               rgb_frame->linesize[0],
100                               QImage::Format_RGB32);
101
102                    emit frameDecoded(img.copy());
103                }
104
105                av_free(buffer);
106                av_frame_free(&rgb_frame);
107            }
108
109            av_packet_unref(&pkt);
110        }
111
112        av_frame_free(&frame);
113        cleanup();
114
115        if(need_reconnect_ && reconnect_attempts_++ < max_reconnect_attempts_) {
116            QThread::msleep(2000);
117        } else {
118            break;
119        }
120    }

124 * @brief 初始化 FFmpeg，打开 RTSP 流并准备解码器
125 * @return 成功返回 true，失败返回 false
126 */
127bool initFFmpeg() {
128    cleanup();
129    avformat_network_init();
130
131    AVDictionary *opts = nullptr;
132    av_dict_set(&opts, "stimeout", "5000000", 0);
133    av_dict_set(&opts, "max_delay", "500000", 0);
134    av_dict_set(&opts, "rtsp_transport", "tcp", 0);
135    av_dict_set(&opts, "fflags", "nobuffer", 0);
136    av_dict_set(&opts, "flags", "low_delay", 0);
137    av_dict_set(&opts, "analyzeduration", "450000", 0);
138    av_dict_set(&opts, "probesize", "96", 0);
139    av_dict_set(&opts, "infbuf", "1", 0);
140
141    if(avformat_open_input(&fmt_ctx_, rtsp_url_.toUtf8().constData(), nullptr, &opts) != 0) {
142        av_dict_free(&opts);
143        return false;
144    }
145    av_dict_free(&opts);
146
147    if(avformat_find_stream_info(fmt_ctx_, nullptr) < 0) return false;
148
149    video_stream_index_ = -1;
150    for(unsigned int i = 0; i < fmt_ctx_->nb_streams; i++) {
151        if(fmt_ctx_->streams[i]->codecpar->codec_type == AVMEDIA_TYPE_VIDEO) {
152            video_stream_index_ = i;
153            break;
154        }
155    }
156    if(video_stream_index_ == -1) return false;
157
158    AVCodecParameters *codecpar = fmt_ctx_->streams[video_stream_index_]->codecpar;
159    AVCodec *codec = avcodec_find_decoder(codecpar->codec_id);
160    if(!codec) return false;
161
162    codec_ctx_ = avcodec_alloc_context3(codec);
163    if(avcodec_parameters_to_context(codec_ctx_, codecpar) < 0) return false;
164
165    if(avcodec_open2(codec_ctx_, codec, nullptr) < 0) return false;
166
167    sws_ctx_ = sws_getContext(codec_ctx_->width, codec_ctx_->height,
168                              codec_ctx_->pix_fmt,
169                              codec_ctx_->width, codec_ctx_->height,
170                              AV_PIX_FMT_RGB32,
171                              SWS_BILINEAR, nullptr, nullptr, nullptr);
172    return sws_ctx_ != nullptr;
173}

 44/**
 45 * @brief 构造函数，初始化扭矩计算参数
 46 */
 47torque::torque() {
 48    R = ((LLA - La) / A) * (0.0085 / 2) + 0.065;        // 初始绕线盘半径
 49    M2 = M1 + (LLA - La) * 0.12;                        // 绕线盘与绳子的总质量
 50    Fa = M3 * g * u3 + Fa_1;                            // 丝杆推力
 51    Ta = (Fa * P) / (2 * PI);                           // 推力与扭矩关系
 52    Tc = (Ta / Zi) * q;                                 // 调整系数后的扭矩
 53}
 54
 55/**
 56 * @brief 根据绳索长度 La_1 计算总扭矩 TG
 57 * @param La_1 当前绳索长度
 58 *
 59 * 计算流程：
 60 * 1. 根据几何关系计算绕线盘角度和坐标
 61 * 2. 计算绳索各段长度与角度
 62 * 3. 计算摩擦力产生的扭矩
 63 * 4. 合成各段力矩得到总扭矩 TG
 64 * 5. 输出百分比形式 TG_1
 65 */
 66void torque::TG_Measure(double La_1) {
 67    La = La_1;
 68
 69    // 绕线盘几何角度计算
 70    double angle_1 = atan(d1 / l1) * 180 / PI;
 71    l2 = sqrt(d1*d1 + l1*l1);
 72    double angle_2 = acos(R / l2) * 180 / PI;
 73    double angle_3 = 90 - (angle_2 + angle_1);
 74    double angle_rad_3 = angle_3 / 180.0 * PI;
 75
 76    // 坐标计算
 77    double Xa = R * sin(angle_rad_3);
 78    double Ya = R * cos(angle_rad_3) + d2;
 79    double Xb = l1;
 80    double Yb = d1 + d2;
 81    double Xc = l1 + l3;
 82    double Yc = d1 + d2 + d3;
 83    double Xd = l1 + l3 + La;
 84    double Yd = 0.17;
 85
 86    // 绳索长度与拉伸计算
 87    l4 = Xd - l1 - l3;
 88    L4 = sqrt(pow((d2+d1+d3-d4),2) + pow(l4,2));
 89
 90    // 更新绕线盘半径
 91    if (L4 < previous_D) {
 92        R += 0.0085 / 2;
 93        DL = 2 * PI * R;
 94        DD = DL * A;
 95        previous_D = L4 - DD;
 96    }
 97
 98    // 绕线盘与绳子重量的摩擦扭矩
 99    double W = M2 * g;
100    double Ff = W * u1;
101    double Tf = Ff * R;
102
103    // 绳索悬空力计算
104    double angle_rad_4 = atan((d1 + d2 + d3 - Yd)/LB);
105    G = LB * 1.2;
106    double F5_1 = G / sin(angle_rad_4);
107    double Ff2 = (l4 - LB) * 0.12 * g * u2;
108    double F5_2 = Ff2 / cos(angle_rad_4);
109    double F5 = F5_1 + F5_2;
110
111    // 合力转矩
112    double angle_rad_5 = atan(d3 / l3);
113    double F2 = F5 * cos(angle_rad_5) / cos(angle_rad_3);
114    TG = F2 * R + Tf + Tc;
115
116    // 百分比表示
117    TG_1 = TG / 1.27 / 20 * 100;
118
119    // 输出调试信息
120    qDebug() << "TG =" << TG;
121    qDebug() << "F2 =" << F2;
122    qDebug() << "F5 =" << F5;
123    qDebug() << "TG_1 =" << TG_1 << "%";
124}

 36/**
 37 * @brief 启动视频录制
 38 * @param filename 输出文件名
 39 * @param width 视频宽度
 40 * @param height 视频高度
 41 * @param fps 帧率
 42 * @return 是否成功启动
 43 */
 44bool startRecording(const QString &filename, int width, int height, int fps)
 45{
 46    if (m_recording) {
 47        emit errorOccurred("Already recording");
 48        return false;
 49    }
 50
 51    m_frameCount = 0; 
 52    m_width = width;
 53    m_height = height;
 54    m_fps = fps;
 55    m_filename = filename;
 56
 57    // 清空队列
 58    {
 59        QMutexLocker locker(&m_mutex);
 60        m_frameQueue.clear();
 61    }
 62
 63    // 初始化输出格式
 64    avformat_alloc_output_context2(&m_formatContext, nullptr, nullptr, filename.toUtf8().constData());
 65    if (!m_formatContext) {
 66        emit errorOccurred("Could not create output context");
 67        return false;
 68    }
 69
 70    AVCodec *codec = avcodec_find_encoder(AV_CODEC_ID_H264);
 71    if (!codec) {
 72        emit errorOccurred("H.264 codec not found");
 73        return false;
 74    }
 75
 76    m_videoStream = avformat_new_stream(m_formatContext, codec);
 77    if (!m_videoStream) {
 78        emit errorOccurred("Could not create video stream");
 79        return false;
 80    }
 81
 82    m_codecContext = avcodec_alloc_context3(codec);
 83    if (!m_codecContext) {
 84        emit errorOccurred("Could not alloc codec context");
 85        return false;
 86    }
 87
 88    // 设置编码器参数
 89    m_codecContext->width = width;
 90    m_codecContext->height = height;
 91    m_codecContext->time_base = {1, fps};
 92    m_codecContext->framerate = {fps, 1};
 93    m_codecContext->pix_fmt = AV_PIX_FMT_YUV420P;
 94    m_codecContext->bit_rate = 4000000;
 95
 96    av_opt_set(m_codecContext->priv_data, "preset", "fast", 0);
 97    av_opt_set(m_codecContext->priv_data, "tune", "zerolatency", 0);
 98
 99    if (avcodec_open2(m_codecContext, codec, nullptr) < 0) {
100        emit errorOccurred("Could not open codec");
101        avcodec_free_context(&m_codecContext);
102        return false;
103    }
104
105    if (avcodec_parameters_from_context(m_videoStream->codecpar, m_codecContext) < 0) {
106        emit errorOccurred("Could not copy codec parameters");
107        avcodec_free_context(&m_codecContext);
108        return false;
109    }
110
111    if (!(m_formatContext->oformat->flags & AVFMT_NOFILE)) {
112        if (avio_open(&m_formatContext->pb, filename.toUtf8().constData(), AVIO_FLAG_WRITE) < 0) {
113            emit errorOccurred("Could not open output file");
114            avcodec_free_context(&m_codecContext);
115            avformat_free_context(m_formatContext);
116            return false;
117        }
118    }
119
120    if (avformat_write_header(m_formatContext, nullptr) < 0) {
121        emit errorOccurred("Could not write header");
122        if (m_formatContext && !(m_formatContext->oformat->flags & AVFMT_NOFILE))
123            avio_closep(&m_formatContext->pb);
124        avcodec_free_context(&m_codecContext);
125        avformat_free_context(m_formatContext);
126        return false;
127    }
128
129    // 初始化 RGB->YUV 转换
130    m_swsContext = sws_getContext(width, height, AV_PIX_FMT_RGB32,
131                                  width, height, AV_PIX_FMT_YUV420P,
132                                  SWS_BILINEAR, nullptr, nullptr, nullptr);
133    if (!m_swsContext) {
134        emit errorOccurred("Could not create SWS context");
135        if (m_formatContext && !(m_formatContext->oformat->flags & AVFMT_NOFILE))
136            avio_closep(&m_formatContext->pb);
137        avcodec_free_context(&m_codecContext);
138        avformat_free_context(m_formatContext);
139        return false;
140    }
141
142    m_stopRequested = false;
143    m_recording = true;
144    m_thread = std::thread(&VideoRecorder::recordingThread, this);
145
146    emit recordingStarted();
147    return true;
148}

153void stopRecording()
154{
155    if (!m_recording) return;
156
157    m_stopRequested = true;
158    m_condition.wakeAll();
159

172void addFrame(const QImage &frame)
173{
174    if (!m_recording) return;
175
176    QImage local = frame;
177    if (local.format() != QImage::Format_RGB32) {
178        local = local.convertToFormat(QImage::Format_RGB32);
179    }
180
181    QMutexLocker locker(&m_mutex);
182    if (m_frameQueue.size() >= MAX_QUEUE_SIZE) {
183        m_frameQueue.takeFirst(); // 队列满则丢弃最早帧
184    }
185    m_frameQueue.append(local.copy());
186    m_condition.wakeOne();
187}

198/**
199 * @brief 录制线程函数
200 *
201 * 功能：
202 * - 从队列取帧
203 * - RGB->YUV420P 转换
204 * - 发送给 H.264 编码器
205 * - 写入输出文件
206 * - 结束时 flush 编码器并写 trailer
207 */
208void recordingThread()
209{
210    AVFrame *yuvFrame = av_frame_alloc();
211    yuvFrame->format = AV_PIX_FMT_YUV420P;
212    yuvFrame->width = m_width;
213    yuvFrame->height = m_height;
214    if (av_frame_get_buffer(yuvFrame, 32) < 0) {
215        qWarning() << "Failed to alloc yuv buffer";
216        av_frame_free(&yuvFrame);
217        return;
218    }
219
220    AVPacket *packet = av_packet_alloc();
221    if (!packet) {
222        av_frame_free(&yuvFrame);
223        return;
224    }
225
226    while (!m_stopRequested) {
227        QImage frame;
228
229        {
230            QMutexLocker locker(&m_mutex);
231            if (m_frameQueue.isEmpty()) {
232                m_condition.wait(&m_mutex, 200); // 队列为空则等待
233                continue;
234            }
235            frame = m_frameQueue.takeFirst();
236        }
237
238        if (frame.isNull()) continue;
239
240        uint8_t *srcData[4] = { const_cast<uint8_t*>(frame.bits()), nullptr, nullptr, nullptr };
241        int srcLinesize[4] = { frame.bytesPerLine(), 0, 0, 0 };
242
243        sws_scale(m_swsContext, srcData, srcLinesize, 0, m_height,
244                  yuvFrame->data, yuvFrame->linesize);
245
246        yuvFrame->pts = m_frameCount++;
247
248        if (avcodec_send_frame(m_codecContext, yuvFrame) < 0) {
249            qDebug() << "Error sending frame to encoder";
250            continue;
251        }
252
253        while (avcodec_receive_packet(m_codecContext, packet) == 0) {
254            av_packet_rescale_ts(packet, m_codecContext->time_base, m_videoStream->time_base);
255            packet->stream_index = m_videoStream->index;
256            av_interleaved_write_frame(m_formatContext, packet);
257            av_packet_unref(packet);
258        }
259    }
260
261    // flush encoder
262    avcodec_send_frame(m_codecContext, nullptr);
263    while (avcodec_receive_packet(m_codecContext, packet) == 0) {
264        av_packet_rescale_ts(packet, m_codecContext->time_base, m_videoStream->time_base);
265        packet->stream_index = m_videoStream->index;
266        av_interleaved_write_frame(m_formatContext, packet);
267        av_packet_unref(packet);
268    }
269
270    av_write_trailer(m_formatContext);
271
272    av_frame_free(&yuvFrame);
273    av_packet_free(&packet);
274
275    if (m_swsContext) {
276        sws_freeContext(m_swsContext);
277        m_swsContext = nullptr;
278    }
279
280    if (m_codecContext) {
281        avcodec_close(m_codecContext);
282        avcodec_free_context(&m_codecContext);
283        m_codecContext = nullptr;
284    }
285
286    if (m_formatContext && !(m_formatContext->oformat->flags & AVFMT_NOFILE)) {
287        avio_closep(&m_formatContext->pb);
288    }
289
290    if (m_formatContext) {
291        avformat_free_context(m_formatContext);
292        m_formatContext = nullptr;
293    }
294
295    // 清空帧队列
296    {
297        QMutexLocker locker(&m_mutex);
298        m_frameQueue.clear();
299    }
300}
