分析检测点并生成 midgoal 点

该类功能：

  • 发布 midgoal 点供导航模块使用

118DetectionAnalyzer::DetectionAnalyzer()
119{
120    odom_sub = nh.subscribe("/cable/odom", 10, &DetectionAnalyzer::odomCallback, this);
121    detection_sub = nh.subscribe("/transformed_detection_info", 10, &DetectionAnalyzer::detectionCallback, this);
122    finalgoal_sub = nh.subscribe("finalgoal", 10, &DetectionAnalyzer::finalgoalCallback, this);
123
124    midgoal_pub = nh.advertise<geometry_msgs::PoseStamped>("midgoal", 10);
125}
126
127void DetectionAnalyzer::odomCallback(const nav_msgs::Odometry::ConstPtr &msg)
128{
129    current_position_x = msg->pose.pose.position.x;
130}
131
132void DetectionAnalyzer::finalgoalCallback(const geometry_msgs::PoseStamped::ConstPtr &msg)
133{
134    final_position_x = msg->pose.position.x;
135}
136
137void DetectionAnalyzer::detectionCallback(const pipeline_robot::detection::ConstPtr &msg)
138{
139    if (msg->point_stamped.point.x > 0.5 && msg->point_stamped.point.x < 3)
140    {
141        float new_detection_x = msg->point_stamped.point.x + current_position_x;
142        if (new_detection_x > final_position_x)
143            return;
144
145        if (!isSimilar(new_detection_x))
146            detections.push_back(new_detection_x);
147    }
148}
149
150void DetectionAnalyzer::publishMidGoal()
151{
152    if (!detections.empty())
153    {
154        auto min_it = std::min_element(detections.begin(), detections.end());
155        float min_detection_x = *min_it;
156
157        if (!isSimilarPublished(min_detection_x))
158        {
159            geometry_msgs::PoseStamped midgoal;
160            midgoal.header.stamp = ros::Time::now();
161            midgoal.header.frame_id = "odom";
162            midgoal.pose.position.x = min_detection_x - 0.5;
163            midgoal.pose.orientation.w = 1.0;
164            midgoal_pub.publish(midgoal);
165
166            published_positions.push_back(min_detection_x);
167        }
168        detections.erase(min_it);
169    }
170}
171
172bool DetectionAnalyzer::isSimilar(float new_detection_x)
173{
174    for (const float &x : detections)
175    {
176        if (std::fabs(x - new_detection_x) < 1)
177            return true;
178    }
179    return false;
180}
181
182bool DetectionAnalyzer::isSimilarPublished(float x)
183{
184    for (const float &published_x : published_positions)
185    {
186        if (std::fabs(published_x - x) < 0.5)
187            return true;
188    }
189    return false;
190}

192/**
193 * @brief 主程序入口
194 * @param argc 命令行参数数量
195 * @param argv 命令行参数数组
196 * @return int 返回 0 表示正常退出
197 *
198 * 功能：
199 *  - 初始化 ROS 节点
200 *  - 创建 DetectionAnalyzer 对象
201 *  - 循环处理回调并周期性发布 midgoal
202 */
203int main(int argc, char **argv)
204{
205    ros::init(argc, argv, "detection_analyzer");
206    DetectionAnalyzer analyzer;
207
208    ros::Rate loop_rate(10);
209
210    while (ros::ok())
211    {
212        ros::spinOnce();
213        analyzer.publishMidGoal();
214        loop_rate.sleep();
215    }
216
217    return 0;
218}

229        }
230    }

  3 * @brief ROS 节点，用于发布机器人导航的中途目标点和最终目标点
  4 *
  5 * 该文件实现了一个 ROS 节点 WaypointPublisher，功能如下：
  6 * - 接收 midgoal 中途目标点并排序发布
  7 * - 接收 finalgoal 最终目标点并发布

  9 * - 发布自动导航停止信号 (/whether_auto_navigation)

 21 * @class WaypointPublisher
 22 * @brief 中途目标点与最终目标点发布器
 23 *
 24 * 功能：
 25 * - 订阅 midgoal、finalgoal 和机器人状态
 26 * - 根据机器人当前位置动态发布 waypoint
 27 * - 管理中途目标队列，并保证按 X 坐标升序发布

 37     * - waypoint: 发布机器人当前目标点
 38     * - /whether_auto_navigation: 发布自动导航停止信号
 39     *
 40     * 初始化订阅者：
 41     * - midgoal: 中途目标点
 42     * - finalgoal: 最终目标点
 43     * - robot_status: 机器人是否到达目标点
 44     * - /cable/odom: 获取机器人当前位置
 45     */
 46    WaypointPublisher()
 47    {
 48        waypoint_pub = nh.advertise<geometry_msgs::PoseStamped>("waypoint", 10);
 49        auto_navigation_stop_pub = nh.advertise<std_msgs::Bool>("/whether_auto_navigation", 10);
 50        midgoal_sub = nh.subscribe("midgoal", 10, &WaypointPublisher::midgoalCallback, this);
 51        finalgoal_sub = nh.subscribe("finalgoal", 10, &WaypointPublisher::finalgoalCallback, this);
 52        status_sub = nh.subscribe("robot_status", 10, &WaypointPublisher::statusCallback, this);
 53        odom_sub = nh.subscribe("/cable/odom", 10, &WaypointPublisher::odomCallback, this);
 54        finalgoal.pose.position.x = std::numeric_limits<float>::max();
 55        pre_goal.pose.position.x = -1;
 56    }
 57
 58    /**
 59     * @brief midgoal 回调函数
 60     * @param msg 接收到的中途目标点 geometry_msgs::PoseStamped 消息
 61     *
 62     * 功能：
 63     * - 将接收到的中途目标点加入队列
 64     * - 输出日志
 65     */
 66    void midgoalCallback(const geometry_msgs::PoseStamped::ConstPtr &msg)
 67    {
 68        midgoals.push(*msg);
 69        ROS_INFO("已加载中途目标点: X:%f,Y:%f", msg->pose.position.x, msg->pose.position.y);
 70    }
 71
 72    /**
 73     * @brief 将队列中的 PoseStamped 按 X 坐标升序排序
 74     * @param msgQueue 待排序队列
 75     * @return 排序后的队列
 76     */
 77    std::queue<geometry_msgs::PoseStamped> queueSort(std::queue<geometry_msgs::PoseStamped> &msgQueue)
 78    {
 79        std::vector<geometry_msgs::PoseStamped> tempVec;
 80
 81        // 将队列中的元素提取到 vector 中
 82        while (!msgQueue.empty()) {
 83            tempVec.push_back(msgQueue.front());
 84            msgQueue.pop();
 85        }
 86
 87        // 对 vector 中的元素进行排序，根据 x 坐标升序排序
 88        std::sort(tempVec.begin(), tempVec.end(), [](const auto& a, const auto& b) {
 89            return a.pose.position.x < b.pose.position.x;
 90        });
 91
 92        // 将排序后的元素重新放回队列中
 93        for (const auto& pose : tempVec) {
 94            msgQueue.push(pose);
 95        }
 96
 97        return msgQueue;
 98    }

100    /**
101     * @brief odom 回调函数
102     * @param msg 接收到的机器人位姿 nav_msgs::Odometry 消息
103     *

105     * - 更新当前机器人位置和速度

108     */
109    void odomCallback(const nav_msgs::Odometry::ConstPtr &msg)
110    {
111        current_position = msg->pose.pose.position.x;
112        current_status = msg->twist.twist.linear.x;
113        if (abs(current_position - finalgoal.pose.position.x) < 0.3)
114        {
115            std_msgs::Bool msg;
116            msg.data = false;
117            finalinital = true;
118            auto_navigation_stop_pub.publish(msg);
119            finalgoal_reached = true;
120        }
121        else
122        {
123            finalgoal_reached = false;
124        }
125    }

129     * @param msg 接收到的最终目标点 geometry_msgs::PoseStamped 消息
130     *
131     * 功能：
132     * - 更新 finalgoal 成员变量
133     * - 标记最终目标已接收
134     * - 如果 finalinital 为 true，则立即发布最终目标点
135     */
136    void finalgoalCallback(const geometry_msgs::PoseStamped::ConstPtr &msg)
137    {
138        finalgoal = *msg;
139        finalgoal_received = true;
140
141        ROS_INFO("已加载最终目标点: X:%f,Y:%f", finalgoal.pose.position.x, finalgoal.pose.position.y);
142
143        if (finalinital)
144        {
145            waypoint_pub.publish(finalgoal);
146            waypoint_pub.publish(finalgoal);
147            waypoint_pub.publish(finalgoal);
148            pre_goal = finalgoal;
149        }
150        finalinital = false;
151    }
152
153    /**
154     * @brief 处理目标点逻辑，动态发布 waypoint
155     *
156     * 功能：
157     * - 根据机器人当前位置和中途目标队列判断当前目标点
158     * - 判断是否到达当前目标点
159     * - 发布 waypoint，确保中途目标按顺序发布
160     * - 到达最终目标时，发布停止信号
161     */
162    void processGoals()
163    {
164        if (!midgoals.empty())
165        {
166            if (midgoals.size() > 1)
167            {
168                queueSort(midgoals);
169            }
170            current_goal = midgoals.front();
171            if (current_goal.pose.position.x < pre_goal.pose.position.x)
172            {
173                ROS_INFO("接收到更近的目标点:X:%f,Y:%f,优先处理", current_goal.pose.position.x, current_goal.pose.position.y);
174                midinital = true;
175                waypoint_pub.publish(current_goal);
176                midgoals.push(pre_goal);
177                pre_goal = current_goal;
178                midgoals.pop();
179            }
180
181            if (goal_reached || midinital)
182            {
183                current_goal = midgoals.front();
184                if (current_goal == finalgoal && (midgoals.size() - 1 > 0))
185                {
186                    midgoals.pop();
187                    current_goal = midgoals.front();
188                }
189
190                if (current_goal != pre_goal)
191                {
192                    midinital = false;
193                    if (pre_goal.pose.position.x == -1)
194                    {
195                        ROS_INFO("发布第一个目标点: X:%f,Y:%f", current_goal.pose.position.x, current_goal.pose.position.y);
196                        waypoint_pub.publish(current_goal);
197                        midgoals.pop();
198                        if (midgoals.empty())
199                        {
200                            if (!finalgoal_received)
201                            {
202                                ROS_INFO("加载最终发布点");
203                            }
204                            ROS_INFO("无中途发布点,队列补充最终发布点1");
205                            midgoals.push(finalgoal);
206                            midinital = true;
207                        }
208                        pre_goal = current_goal;
209                    }
210                    else if (abs(current_position - pre_goal.pose.position.x) < 0.05)
211                    {
212                        ROS_INFO("目标点: X:%f,Y:%f已到达,停留10s", pre_goal.pose.position.x, pre_goal.pose.position.y);
213                        ros::Duration(10).sleep();
214                        ROS_INFO("发布下一个目标点: X:%f,Y:%f", current_goal.pose.position.x, current_goal.pose.position.y);
215                        waypoint_pub.publish(current_goal);
216                        midgoals.pop();
217                        if (midgoals.empty() && finalgoal_received)
218                        {
219                            ROS_INFO("无中途发布点,队列补充最终发布点2");
220                            midgoals.push(finalgoal);
221                            midinital = true;
222                        }
223                        pre_goal = current_goal;
224                    }
225                    else
226                    {
227                        float dis = abs(current_position - pre_goal.pose.position.x);
228                        ROS_INFO("正在前往目标点: X:%f,Y:%f,距离为:%f", pre_goal.pose.position.x, pre_goal.pose.position.y, dis);
229                    }
230                }
231            }
232        }
233    }

235    /**
236     * @brief 状态回调函数
237     * @param msg 接收到的机器人状态 std_msgs::Bool 消息
238     *
239     * 功能：
240     * - 根据消息更新 goal_reached 标志
241     */
242    void statusCallback(const std_msgs::Bool::ConstPtr &msg)
243    {
244        if (msg->data)
245        {
246            goal_reached = true;
247        }
248        else
249        {
250            goal_reached = false;
251        }
252    }

255    ros::NodeHandle nh;                                 ///< ROS 节点句柄
256    ros::Publisher waypoint_pub, auto_navigation_stop_pub; ///< 发布 waypoint 和自动导航停止信号
257    ros::Subscriber midgoal_sub, finalgoal_sub, odom_sub; ///< 订阅中途目标、最终目标和 odom

260    geometry_msgs::PoseStamped current_goal;            ///< 当前目标点
261    geometry_msgs::PoseStamped finalgoal;               ///< 最终目标点

266    bool goal_reached = false;                          ///< 是否到达当前目标
267    bool finalgoal_reached = false;                     ///< 是否到达最终目标
268    bool midinital = true;                              ///< 是否处理中途目标初始化
269    bool finalinital = true;                            ///< 是否最终目标初始化
