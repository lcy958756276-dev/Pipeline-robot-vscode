 9 * @brief Main entry point for RTSP reader node

12 * @param argv Array of command line arguments

18int main(int argc, char** argv) {
19    ros::init(argc, argv, "rtsp_reader_node"); // 初始化 ROS 节点
20    ros::NodeHandle nh;
21
22    // 发布自定义 RGBD 压缩消息
23    ros::Publisher rgbd_pub = nh.advertise<pipeline_robot::compressedRGBD>("/camera/rgb_depth/compressed", 1);
24
25    std::string rtsp_url = "rtsp://admin:123456@192.168.8.10/Streaming/Channels/2"; // RTSP 视频流 URL
26    cv::VideoCapture cap(rtsp_url);
27
28    if (!cap.isOpened()) {
29        ROS_ERROR("Failed to open RTSP stream: %s", rtsp_url.c_str());
30        return -1; // 如果无法打开 RTSP 流，返回错误
31    } else {
32        ROS_INFO("Successfully connected to RTSP stream: %s", rtsp_url.c_str());
33    }
34
35    ros::Rate loop_rate(30); // 设定循环频率为 30 FPS
36    int frame_count = 0; // 帧计数器
37
38    while (ros::ok()) {
39        cv::Mat frame;
40        if (!cap.read(frame)) {
41            ROS_WARN("Failed to read frame from RTSP stream, retrying...");
42            cap.release(); // 释放当前的流
43            ros::Duration(1.0).sleep(); // 等待 1 秒后重新连接
44            cap.open(rtsp_url); // 重新打开 RTSP 流
45            if (cap.isOpened()) {
46                ROS_INFO("Reconnected to RTSP stream successfully.");
47            } else {
48                ROS_ERROR("Reconnect failed. Will retry...");
49            }
50            continue; // 如果读取失败，则继续循环
51        }
52
53        frame_count++; // 增加帧计数
54        //ROS_INFO("Received frame #%d, size: %dx%d", frame_count, frame.cols, frame.rows);
55
56        // --- 压缩 RGB 图像 ---
57        std::vector<uchar> buf; // 存储压缩后的图像数据
58        std::vector<int> params = {cv::IMWRITE_JPEG_QUALITY, 75}; // 压缩质量设置为 75
59        cv::imencode(".jpg", frame, buf, params); // 压缩图像为 JPEG 格式
60
61        // 构建自定义 RGBD 消息
62        pipeline_robot::compressedRGBD msg;
63        msg.rgb.header.stamp = ros::Time::now(); // 当前时间戳
64        msg.rgb.header.frame_id = "camera"; // 设置帧 ID
65        msg.rgb.format = "jpeg"; // 图像格式
66        msg.rgb.data.assign(buf.begin(), buf.end()); // 将压缩图像数据赋值给消息
67
68        // 如果 depth 数据有，也可以在这里填充 msg.depth
69        // msg.depth = ...
70
71        rgbd_pub.publish(msg); // 发布消息
72
73        ros::spinOnce(); // 处理回调函数
74        loop_rate.sleep(); // 按照设定的频率睡眠
75    }
76
77    cap.release(); // 释放视频流
78    return 0; // 正常结束
79}
80} // namespace cameralcapture
