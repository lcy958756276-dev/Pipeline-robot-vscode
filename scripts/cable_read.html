  1/**
  2 * @file odometry_publisher.cpp
  3 * @brief 基于轮速编码器计算里程计并发布 /cable/odom，同时广播 tf 变换
  4 *
  5 * 功能：
  6 * - 订阅 `/minipc_echo` 获取轮速编码器脉冲。
  7 * - 根据轮半径和脉冲计算机器人行进距离。
  8 * - 发布 `/cable/odom` 里程计消息。
  9 * - 广播 `odom -> base_link` 的 tf 变换。
 10 * - 支持通过 `/reset_odom` 重置里程计。
 11 */
 12
 13#include <ros/ros.h>
 14#include <std_msgs/Int32.h>
 15#include <nav_msgs/Odometry.h>
 16#include <tf/transform_broadcaster.h>
 17#include "std_msgs/String.h"
 18#include <regex>
 19#include <std_msgs/Bool.h>
 20
 21/** @brief 全局变量 */
 22
 23// 重置里程计标志
 24bool reset_odom = false;
 25
 26// 轮子参数
 27double wheel_radius = 0.0318;  ///< 轮子半径(m)
 28double wheel_base = 0.343;     ///< 轮距(m)
 29
 30// 行进距离
 31double distance = 0.0;
 32// double right_distance = 0.0; // 右轮距离（未使用）
 33
 34// 当前位姿
 35double x = 0.0;
 36double y = 0.0;
 37double theta = 0.0;
 38
 39// 每脉冲对应的角度
 40double pluse = 2.295048561 * 360.0 / (400*4);
 41
 42// 上一次更新时间
 43ros::Time last_time;
 44
 45/**
 46 * @brief 从编码器字符串消息中提取左轮增量数
 47 * @param input 编码器字符串消息
 48 * @return std::string 左轮增量字符串
 49 */
 50std::string extra(const std::string& input)
 51{
 52    std::string prefix="e1:";
 53    std::string num;
 54    std::size_t prefix_pos = input.find(prefix);
 55    for(int i = prefix_pos+3; i < input.size(); i++)
 56    {
 57        if(input[i] == 'p') break;
 58        num += input[i];
 59    }
 60    return num;
 61}

 65 * @param msg std_msgs::Bool 消息，true 表示重置

 67void resetOdometryCallback(const std_msgs::Bool::ConstPtr& msg)
 68{
 69    if (msg->data == true)
 70    {
 71        reset_odom = true; // 设置重置标志为真
 72    }
 73}

 75/**
 76 * @brief 接收编码器字符串消息，计算左轮行进距离
 77 * @param msg std_msgs::String 消息
 78 */
 79void EncoderCallback(const std_msgs::String::ConstPtr &msg)
 80{
 81    std::string str = msg->data;
 82    if (str.find("e1") == std::string::npos)
 83    {
 84        return;
 85    }
 86    
 87    std::string num_str = extra(str);
 88    if(num_str.size() > 0)
 89    {
 90        int left_increment = std::stoi(num_str);
 91        // 根据脉冲数计算左轮行进距离
 92        distance += (left_increment * (pluse / 360.0) * 2 * M_PI * wheel_radius);
 93    }
 94}
 95
 96/**
 97 * @brief 发布里程计消息并广播 tf 变换
 98 * @param odom_pub ROS 发布器，发布 nav_msgs::Odometry
 99 * @param tf_broadcaster tf 广播器
100 */
101void publishOdometry(ros::Publisher& odom_pub, tf::TransformBroadcaster& tf_broadcaster)
102{
103    ros::Time current_time = ros::Time::now();
104    double dt = (current_time - last_time).toSec();
105
106    // 处理里程计重置
107    if (reset_odom)
108    {
109        x = 0.0;
110        y = 0.0;
111        theta = 0.0;
112        reset_odom = false; // 重置完成
113    }
114
115    double d_distance = distance; // 计算行进距离
116    x += d_distance;              // 更新 x 坐标
117    // y += d_distance * sin(theta); // 更新 y 坐标（未使用）
118    theta = 0;                     // 更新角度
119
120    // 创建里程计消息
121    nav_msgs::Odometry odom;
122    odom.header.frame_id = "odom";
123    odom.pose.pose.position.x = x;
124    odom.pose.pose.position.y = 0;
125    odom.pose.pose.position.z = 0.0;
126    odom.pose.pose.orientation = tf::createQuaternionMsgFromYaw(theta);
127    odom.child_frame_id = "base_link";
128    odom.twist.twist.linear.x = d_distance / dt;
129
130    // 发布里程计
131    odom_pub.publish(odom);
132
133    // 广播 tf 变换
134    tf::Transform transform;
135    transform.setOrigin(tf::Vector3(x, y, 0.0));
136    tf::Quaternion q;
137    q.setRPY(0, 0, theta);
138    transform.setRotation(q);
139    tf_broadcaster.sendTransform(tf::StampedTransform(transform, ros::Time::now(), "odom", "base_link"));
140
141    // 重置行进距离
142    distance = 0.0;
143    last_time = current_time;
144}

146/**
147 * @brief 主函数，初始化 ROS 节点，订阅编码器和重置指令，发布里程计
148 * @param argc 命令行参数数量
149 * @param argv 命令行参数数组
150 * @return int 程序退出状态
151 */
152int main(int argc, char** argv)
153{
154    ros::init(argc, argv, "odometry_publisher");
155    ros::NodeHandle nh;
156
157    // 订阅编码器数据
158    ros::Subscriber ticks_sub = nh.subscribe("/minipc_echo", 1000, EncoderCallback);
159    // 订阅重置指令
160    ros::Subscriber reset_sub = nh.subscribe("/reset_odom", 10, resetOdometryCallback);
161
162    // 发布里程计
163    ros::Publisher odom_pub = nh.advertise<nav_msgs::Odometry>("/cable/odom", 1);
164
165    // tf 广播器
166    tf::TransformBroadcaster tf_broadcaster;
167
168    // 初始化循环频率
169    ros::Rate loop_rate(10); // 10Hz
170
171    while (ros::ok())
172    {
173        // 发布里程计消息和 tf
174        publishOdometry(odom_pub, tf_broadcaster);
175
176        ros::spinOnce();
177        loop_rate.sleep();
178    }
179
180    return 0;
181}

  1/**
  2 * @file serial_readwrite_node.cpp
  3 * @brief ROS节点，用于通过串口与STM32通信，实现数据接收和发送
  4 *
  5 * 功能：
  6 * - 订阅 "cable" 主题，将接收到的字符串通过串口发送给STM32。
  7 * - 接收STM32返回的数据，发布到 "minipc_echo" 主题。
  8 * - 支持对特定命令（如 "AC" 开头）进行数值处理后再发送。
  9 */
 10
 11#include <ros/ros.h>
 12#include <serial/serial.h> // ROS 内置串口库
 13#include <std_msgs/String.h>
 14#include <std_msgs/Empty.h>
 15#include "std_msgs/UInt8.h"
 16#include "std_msgs/UInt8MultiArray.h"
 17#include <cstdlib>
 18#include <string>
 19#include <stdexcept>
 20#include <cmath>
 21
 22/** @brief 全局串口对象 */
 23serial::Serial ser;
 24
 25/**
 26 * @brief 处理订阅到的发送数据，将其发送到串口
 27 * @param msg std_msgs::String 发送的字符串消息
 28 *
 29 * 功能：
 30 * - 如果消息以 "AC" 开头，将消息中的数字解析出来，进行特定计算后发送。
 31 * - 否则直接将消息追加 "\r\n" 发送到串口。
 32 */
 33void datasend_callback(const std_msgs::String::ConstPtr &msg)
 34{
 35    int len = msg->data.size();
 36    ROS_INFO("Successfully get: %s", msg->data.c_str());
 37
 38    // 特殊处理 "AC" 开头的消息
 39    if (msg->data.substr(0, 2) == "AC")
 40    {
 41        double number = std::stod(msg->data.substr(3));
 42        // 根据公式处理数字
 43        double newNumber = 72000000 / (number * 1200 * 3000 / 0.17) - 1;
 44        std::string output = "AC+" + std::to_string(newNumber);
 45        std::string command_to_send = output + "\r\n";
 46        ROS_INFO("Successfully send: %s", command_to_send.c_str());
 47        ser.write(command_to_send);
 48        return;
 49    }
 50
 51    // 普通字符串直接发送
 52    std::string command_to_send = msg->data + "\r\n";
 53    ser.write(command_to_send);
 54}
 55
 56/**
 57 * @brief 主函数，初始化节点，打开串口，订阅/发布主题
 58 * @param argc 命令行参数个数
 59 * @param argv 命令行参数数组
 60 * @return int 程序退出状态
 61 *
 62 * 功能：
 63 * - 初始化 ROS 节点。
 64 * - 打开串口 "/dev/ttyACM0"，波特率 19200。
 65 * - 订阅 "cable" 主题发送数据。
 66 * - 发布 "minipc_echo" 主题接收数据。
 67 * - 循环读取串口数据并发布。
 68 */
 69int main(int argc, char **argv)
 70{
 71    // 初始化节点
 72    ros::init(argc, argv, "serial_readwrite_node");
 73    ros::NodeHandle nh;
 74
 75    static int len;
 76    std_msgs::String RecvData;
 77
 78    // 订阅发送主题
 79    ros::Subscriber send_sub = nh.subscribe("cable", 1000, datasend_callback);
 80    // 发布接收主题
 81    ros::Publisher read_pub = nh.advertise<std_msgs::String>("minipc_echo", 1000);
 82
 83    try
 84    {
 85        // 设置串口属性并打开
 86        ser.setPort("/dev/ttyACM0");
 87        ser.setBaudrate(19200);
 88        serial::Timeout to = serial::Timeout::simpleTimeout(1000);
 89        ser.setTimeout(to);
 90        ser.open();
 91
 92        // 使用 stty 设置串口参数
 93        std::string port = "/dev/ttyACM0";
 94        std::string command = "stty -F " + port + " cs8 -cstopb -parenb";
 95        if (system(command.c_str()) != 0)
 96        {
 97            ROS_ERROR_STREAM("Failed to set serial port parameters using stty");
 98            return -1;
 99        }
100    }
101    catch (serial::IOException &e)
102    {
103        ROS_ERROR_STREAM("Unable to open port /dev/ttyACM0");
104        return -1;
105    }
106
107    if (ser.isOpen())
108    {
109        ROS_INFO_STREAM("Serial Port initialized");
110    }
111    else
112    {
113        return -1;
114    }
115
116    ros::Rate loop_rate(10); // 循环频率 10Hz
117
118    while (ros::ok())
119    {
120        len = ser.available();
121
122        if (len > 2)
123        {
124            std_msgs::UInt8MultiArray serial_data;
125
126            // 读取串口数据
127            ser.read(serial_data.data, len);
128
129            RecvData.data.clear(); // 清除上一次数据
130
131            for (size_t i = 0; i < len; i++)
132            {
133                RecvData.data.push_back(serial_data.data[i]);
134            }
135
136            ROS_INFO("Successfully received: %s", RecvData.data.c_str());
137
138            // 发布接收到的数据
139            if (RecvData.data.size() > 1)
140            {
141                read_pub.publish(RecvData);
142            }
143        }
144
145        // 处理订阅消息
146        ros::spinOnce();
147        loop_rate.sleep();
148    }
149
150    return 0;
151}
