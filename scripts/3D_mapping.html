  1/**
  2 * @file octree_map_builder.cpp
  3 * @brief 使用八叉树对累计点云进行存储与管理，同时根据机器人位姿进行点云变换和统计。
  4 * 
  5 * 功能：
  6 * - 接收机器人里程计信息，记录当前位置。
  7 * - 接收点云信息，并根据里程计变换至 odom 坐标系。
  8 * - 使用 pcl::octree::OctreePointCloudVoxelCentroid 构建八叉树。
  9 * - 累计统计点云数量和八叉树体素数量。
 10 * - 发布整体地图点云。
 11 */
 12
 13#include <ros/ros.h>
 14#include <sensor_msgs/PointCloud2.h>
 15#include <nav_msgs/Odometry.h>
 16#include <pcl_conversions/pcl_conversions.h>
 17#include <pcl/point_cloud.h>
 18#include <pcl/point_types.h>
 19#include <pcl_ros/transforms.h>
 20#include <tf/transform_listener.h>
 21#include <pcl/octree/octree_pointcloud_voxelcentroid.h>
 22#include <Eigen/Dense>
 23
 24/** @brief 八叉树分辨率 */
 25float resolution = 0.01f;
 26
 27/** @brief 八叉树对象，用于存储体素中心点 */
 28pcl::octree::OctreePointCloudVoxelCentroid<pcl::PointXYZ> octree(resolution);
 29
 30/** @brief TF 监听器指针，用于坐标变换 */
 31tf::TransformListener* tfListener;
 32
 33// 统计相关变量
 34size_t cumulativePoints = 0;        ///< 累计接收点数
 35size_t currentVoxels = 0;           ///< 当前体素数量
 36Eigen::Vector3d lastPosition = Eigen::Vector3d::Zero();  ///< 上一时刻机器人位置
 37Eigen::Vector3d currentPosition = Eigen::Vector3d::Zero(); ///< 当前机器人位置
 38bool positionInitialized = false;   ///< 位置是否初始化标志
 39
 40/**
 41 * @brief odometry 回调函数，更新机器人当前位置
 42 * 
 43 * @param msg 当前机器人里程计信息（nav_msgs::Odometry）
 44 * @return void
 45 */
 46void odomCallback(const nav_msgs::Odometry::ConstPtr& msg)
 47{
 48    currentPosition = Eigen::Vector3d(
 49        msg->pose.pose.position.x,
 50        msg->pose.pose.position.y,
 51        msg->pose.pose.position.z);
 52
 53    if(!positionInitialized) {
 54        lastPosition = currentPosition;
 55        positionInitialized = true;
 56    }
 57}
 58
 59/**
 60 * @brief 点云回调函数，将接收到的点云变换到 odom 坐标系并更新八叉树
 61 * 
 62 * @param cloudMsg 接收到的点云（sensor_msgs::PointCloud2）
 63 * @return void
 64 */
 65void cloudCallback(const sensor_msgs::PointCloud2::ConstPtr& cloudMsg)
 66{
 67    pcl::PointCloud<pcl::PointXYZ>::Ptr cloud(new pcl::PointCloud<pcl::PointXYZ>);
 68    pcl::fromROSMsg(*cloudMsg, *cloud);
 69
 70    double distanceMoved = (currentPosition - lastPosition).norm();
 71    
 72    if(distanceMoved > 0.01)
 73    {
 74        lastPosition = currentPosition;
 75
 76        pcl::PointCloud<pcl::PointXYZ>::Ptr transformedCloud(new pcl::PointCloud<pcl::PointXYZ>);
 77        if(tfListener->waitForTransform("odom", cloudMsg->header.frame_id, 
 78                                      cloudMsg->header.stamp, ros::Duration(1.0)))
 79        {
 80            pcl_ros::transformPointCloud("odom", *cloud, *transformedCloud, *tfListener);
 81            
 82            // 更新统计信息
 83            cumulativePoints += transformedCloud->size();
 84            
 85            // 更新八叉树
 86            octree.setInputCloud(transformedCloud);
 87            octree.addPointsFromInputCloud();
 88        }
 89    }
 90}
 91
 92/**
 93 * @brief 主函数，初始化 ROS 节点、订阅者、发布者并循环处理点云
 94 * 
 95 * @param argc 命令行参数数量
 96 * @param argv 命令行参数数组
 97 * @return int 程序退出状态
 98 */
 99int main(int argc, char** argv)
100{
101    ros::init(argc, argv, "octree_map_builder");
102    ros::NodeHandle nh;
103    
104    tfListener = new tf::TransformListener();
105
106    ros::Subscriber sub = nh.subscribe("/point_cloud", 10, cloudCallback);
107    ros::Subscriber odomSub = nh.subscribe("/cable/odom", 10, odomCallback);
108    ros::Publisher pub = nh.advertise<sensor_msgs::PointCloud2>("/overall_map_cloud", 1);
109
110    ros::Rate rate(10);
111    
112    while(ros::ok())
113    {
114        ros::spinOnce();
115        
116        // 获取当前八叉树状态
117        currentVoxels = octree.getLeafCount();
118        
119        // 计算内存占用（假设每个点占12字节：3个float）
120        double rawMemoryMB = (cumulativePoints * sizeof(pcl::PointXYZ)) / (1024.0 * 1024.0);
121        double octreeMemoryMB = (currentVoxels * sizeof(pcl::PointXYZ)) / (1024.0 * 1024.0);
122        
123        // 发布点云
124        pcl::PointCloud<pcl::PointXYZ>::Ptr outputCloud(new pcl::PointCloud<pcl::PointXYZ>);
125        octree.getOccupiedVoxelCenters(outputCloud->points);
126        
127        sensor_msgs::PointCloud2 outputMsg;
128        pcl::toROSMsg(*outputCloud, outputMsg);
129        outputMsg.header.stamp = ros::Time::now();
130        outputMsg.header.frame_id = "odom";
131        pub.publish(outputMsg);
132        
133        // 输出统计信息
134        // ROS_INFO_STREAM("Accumulated Points: " << cumulativePoints 
135        //                << " (" << rawMemoryMB << " MB) | "
136        //                << "Octree Voxels: " << currentVoxels 
137        //                << " (" << octreeMemoryMB << " MB)");
138        
139        rate.sleep();
140    }
141    
142    delete tfListener;
143    return 0;
144}

  1/**
  2 * @file da_octree_node.cpp
  3 * @brief 通过订阅 Distance-Angle 数据和外部 odom，生成稀疏化点云并构建八叉树
  4 *
  5 * 功能：
  6 * - 订阅 `/distance_angle` 消息，将距离角度数据转换为点云。
  7 * - 订阅 `/cable/odom` 获取机器人位姿。
  8 * - 使用 pcl::octree::OctreePointCloudVoxelCentroid 对点云进行稀疏化。
  9 * - 发布整体稀疏化点云 `/overall_map_cloud`。
 10 */
 11
 12#include <ros/ros.h>
 13#include <sensor_msgs/PointCloud2.h>
 14#include <nav_msgs/Odometry.h>
 15#include <pcl/point_cloud.h>
 16#include <pcl/point_types.h>
 17#include <pcl_conversions/pcl_conversions.h>
 18#include <pcl/octree/octree_pointcloud_voxelcentroid.h>
 19#include <pipeline_robot/DistanceAngle.h>
 20
 21/**
 22 * @class DA_Octree_Node
 23 * @brief 通过 Distance-Angle 数据生成点云并构建八叉树的节点类
 24 */
 25class DA_Octree_Node {
 26public:
 27    /**
 28     * @brief 构造函数，初始化订阅者、发布者和八叉树
 29     */
 30    DA_Octree_Node() : octree_(0.000001f), odom_received_(false) {
 31        ros::NodeHandle nh;
 32
 33        // 订阅 Distance-Angle
 34        da_sub_ = nh.subscribe("/distance_angle", 1000, &DA_Octree_Node::daCallback, this);
 35
 36        // 订阅外部 odom
 37        odom_sub_ = nh.subscribe("/cable/odom", 10, &DA_Octree_Node::odomCallback, this);
 38
 39        // 发布稀疏化点云
 40        cloud_pub_ = nh.advertise<sensor_msgs::PointCloud2>("/overall_map_cloud", 1);
 41
 42        cloud_.reset(new pcl::PointCloud<pcl::PointXYZ>);
 43        cloud_->is_dense = false;
 44    }
 45
 46    /**
 47     * @brief odom 回调函数，更新当前机器人位姿
 48     * @param msg 当前里程计消息（nav_msgs::Odometry）
 49     */
 50    void odomCallback(const nav_msgs::Odometry::ConstPtr& msg) {
 51        current_pose_ = msg->pose.pose;
 52        odom_received_ = true;
 53    }
 54
 55    /**
 56     * @brief Distance-Angle 回调函数，将距离角度数据转换为点并更新八叉树
 57     * @param msg Distance-Angle 消息（pipeline_robot::DistanceAngle）
 58     */
 59    void daCallback(const pipeline_robot::DistanceAngle::ConstPtr& msg) {
 60        if (!odom_received_) {
 61            ROS_WARN_THROTTLE(5, "Waiting for /cable/odom ...");
 62            return;
 63        }
 64
 65        ros::Time current_time = ros::Time::now();
 66        float angle_rad = msg->angle * M_PI / 180.0;
 67
 68        pcl::PointXYZ point;
 69        point.x = current_pose_.position.x;            ///< 使用外部 odom 的 X
 70        point.y = msg->distance * sin(angle_rad) * 0.01;
 71        point.z = msg->distance * cos(angle_rad) * (-0.01);
 72
 73        cloud_->points.push_back(point);
 74
 75        // 更新八叉树
 76        octree_.setInputCloud(cloud_);
 77        octree_.addPointsFromInputCloud();
 78
 79        pcl::PointCloud<pcl::PointXYZ>::Ptr outputCloud(new pcl::PointCloud<pcl::PointXYZ>);
 80        octree_.getOccupiedVoxelCenters(outputCloud->points);
 81
 82        sensor_msgs::PointCloud2 outputMsg;
 83        pcl::toROSMsg(*outputCloud, outputMsg);
 84        outputMsg.header.stamp = current_time;
 85        outputMsg.header.frame_id = "odom";  ///< 与外部 odom 保持一致
 86        cloud_pub_.publish(outputMsg);
 87
 88        ROS_INFO_STREAM("Octree voxels: " << octree_.getLeafCount()
 89                        << " | Input points: " << cloud_->size());
 90    }
 91
 92private:
 93    ros::Subscriber da_sub_;        ///< Distance-Angle 数据订阅者
 94    ros::Subscriber odom_sub_;      ///< 外部 odom 数据订阅者
 95    ros::Publisher cloud_pub_;      ///< 发布稀疏化点云
 96
 97    pcl::PointCloud<pcl::PointXYZ>::Ptr cloud_; ///< 原始点云存储
 98    pcl::octree::OctreePointCloudVoxelCentroid<pcl::PointXYZ> octree_; ///< 八叉树对象
 99
100    geometry_msgs::Pose current_pose_; ///< 当前机器人位姿
101    bool odom_received_;               ///< odom 是否已接收标志
102};
103
104/**
105 * @brief 主函数，初始化 ROS 节点并启动 DA_Octree_Node
106 * @param argc 命令行参数数量
107 * @param argv 命令行参数数组
108 * @return int 程序退出状态
109 */
110int main(int argc, char** argv) {
111    ros::init(argc, argv, "da_octree_node");
112    DA_Octree_Node node;
113    ros::spin();
114    return 0;
115}
