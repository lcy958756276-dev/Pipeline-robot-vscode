  2 * @file tcp_current_node.cpp
  3 * @brief ROS节点，通过TCP连接接收电流数据，并将解析后的数据发布为Float32消息

  5 * 功能：
  6 * - 通过TCP客户端连接指定IP和端口，接收传感器/控制器发送的电流数据。
  7 * - 解析数据中的 c0~c4 五个通道，并按公式转换为实际电流值。
  8 * - 发布电流值到对应 ROS 话题。

 23/**
 24 * @brief 从输入字符串中提取指定前缀后的数值
 25 * @param input 输入字符串
 26 * @param prefix 前缀，例如 "c0:"
 27 * @return std::string 提取的数值字符串，如果未找到返回空字符串
 28 */
 29std::string extractData(const std::string& input, const std::string& prefix) {
 30    std::size_t pos = input.find(prefix);
 31    if (pos == std::string::npos) return "";
 32    std::string num;
 33    for (std::size_t i = pos + prefix.size(); i < input.size(); ++i) {
 34        if (input[i] == 'c' || input[i] == ' ' || input[i] == '\n') break;
 35        num += input[i];
 36    }
 37    return num;
 38}

 43 * @return true 是整数
 44 * @return false 否

 46bool isValidInteger(const std::string& str) {
 47    if (str.empty()) return false;
 48    char *p;
 49    strtol(str.c_str(), &p, 10);

 54 * @brief 解析一行TCP数据并发布到对应ROS话题

104/**
105 * @brief TCP客户端主循环
106 * @param ip 服务器IP
107 * @param port 服务器端口
108 *
109 * 功能：
110 * - 创建TCP socket并连接服务器。
111 * - 循环接收数据，按行解析。
112 * - 对不完整行进行缓存处理，保证数据完整。
113 * - 调用 parseAndPublish 将解析结果发布到 ROS。
114 */
115void tcpLoop(const std::string& ip, int port) {
116    int sockfd = socket(AF_INET, SOCK_STREAM, 0);
117    if (sockfd < 0) {
118        ROS_ERROR("Socket creation failed");
119        return;
120    }
121
122    sockaddr_in serv_addr;
123    serv_addr.sin_family = AF_INET;
124    serv_addr.sin_port = htons(port);
125    inet_pton(AF_INET, ip.c_str(), &serv_addr.sin_addr);
126
127    if (connect(sockfd, (struct sockaddr*)&serv_addr, sizeof(serv_addr)) < 0) {
128        ROS_ERROR("Connection failed to %s:%d", ip.c_str(), port);
129        return;
130    }
131
132    ROS_INFO("Connected to server %s:%d", ip.c_str(), port);
133
134    char buffer[1024];
135    std::string leftover;
136
137    while (ros::ok()) {
138        ssize_t n = read(sockfd, buffer, sizeof(buffer) - 1);
139        if (n <= 0) {
140            ROS_WARN("Server disconnected");
141            break;
142        }
143        buffer[n] = '\0';
144        std::string data = leftover + buffer;
145
146        std::istringstream iss(data);
147        std::string line;
148        while (std::getline(iss, line)) {
149            if (!line.empty()) parseAndPublish(line);
150        }
151
152        // 处理不完整的一行
153        if (data.back() != '\n')
154            leftover = data.substr(data.find_last_of('\n') + 1);
155        else
156            leftover.clear();
157
158        ros::spinOnce();
159    }
160    close(sockfd);
161}

158/**
159 * @brief TCP 数据接收循环
160 *  从 TCP 套接字接收数据，处理缓存，发布原始数据，同时调用解析函数
161 *  @param raw_pub 发布原始字符串数据
162 *  @param point_pub 发布解析后的 XYZ 坐标点
163 *  @param da_pub 发布解析后的 Distance-Angle 消息
164 */
165void tcpDataLoop(ros::Publisher& raw_pub, 
166                ros::Publisher& point_pub,
167                ros::Publisher& da_pub) {
168    char buffer[4096];
169    std::string remaining_data;
170
171    while (ros::ok()) {
172        ssize_t valread = read(g_sockfd, buffer, sizeof(buffer) - 1);
173        if (valread <= 0) {
174            ROS_WARN_STREAM("Read error: " << (valread == 0 ? "EOF" : strerror(errno)));
175            break;
176        }
177
178        buffer[valread] = '\0';
179        std::string raw_data = remaining_data + buffer;
180        size_t last_newline = raw_data.find_last_of('\n');
181
182        if (last_newline != std::string::npos) {
183            // 发布原始数据
184            std_msgs::String raw_msg;
185            raw_msg.data = "[RAW] " + raw_data.substr(0, 100);  // 截断前100字符
186            raw_pub.publish(raw_msg);
187
188            // 解析有效数据包
189            parseAndPublishData(raw_data.substr(0, last_newline), point_pub, da_pub);
190            remaining_data = raw_data.substr(last_newline + 1);
191        } else {
192            remaining_data = raw_data;
193            ROS_DEBUG("Buffering incomplete packet");
194        }
195        memset(buffer, 0, sizeof(buffer));
196    }
197}

199/**
200 * @brief 解析原始数据并发布 ROS 消息
201 *  解析 XYZ 坐标和 Distance-Angle 数据，并分别发布到对应话题
202 *  @param data 原始数据字符串
203 *  @param point_pub geometry_msgs::PointStamped 发布器
204 *  @param da_pub pipeline_robot::DistanceAngle 发布器
205 */
206void parseAndPublishData(const std::string& data, 
207                        ros::Publisher& point_pub,
208                        ros::Publisher& da_pub) {
209    std::istringstream iss(data);
210    std::string line;
211    geometry_msgs::PointStamped point;
212    point.header.frame_id = "sensor_frame";
213    bool has_coordinates = false;
214
215    pipeline_robot::DistanceAngle da_msg;
216    da_msg.header.stamp = ros::Time::now();
217
218    while (std::getline(iss, line)) {
219        if (line.empty()) continue;
220        ROS_DEBUG_STREAM("Processing: " << line);
221
222        try {
223            // 解析XYZ坐标
224            if (line.find("X=") == 0) {
225                point.point.x = std::stof(line.substr(2));

 78/**
 79 * @brief TCP 接收循环
 80 * 
 81 * 从指定 IP:端口接收数据，每次读取后按行解析
 82 * 并调用 parseAndPublish 处理。处理不完整行缓存。
 83 * 
 84 * @param ip TCP 服务器 IP
 85 * @param port TCP 服务器端口
 86 */
 87void tcpLoop(const std::string& ip, int port) {
 88    int sockfd = socket(AF_INET, SOCK_STREAM, 0);
 89    if (sockfd < 0) {
 90        ROS_ERROR("Socket creation failed");
 91        return;
 92    }
 93
 94    sockaddr_in serv_addr;
 95    serv_addr.sin_family = AF_INET;
 96    serv_addr.sin_port = htons(port);
 97    inet_pton(AF_INET, ip.c_str(), &serv_addr.sin_addr);
 98
 99    if (connect(sockfd, (struct sockaddr*)&serv_addr, sizeof(serv_addr)) < 0) {
100        ROS_ERROR("Connection failed to %s:%d", ip.c_str(), port);
101        return;
102    }
103
104    ROS_INFO("Connected to server %s:%d", ip.c_str(), port);
105
106    char buffer[1024];
107    std::string leftover;
108
109    while (ros::ok()) {
110        ssize_t n = read(sockfd, buffer, sizeof(buffer) - 1);
111        if (n <= 0) {
112            ROS_WARN("Server disconnected");
113            break;
114        }
115        buffer[n] = '\0';
116        std::string data = leftover + buffer;
117
118        std::istringstream iss(data);
119        std::string line;
120        while (std::getline(iss, line)) {
121            if (!line.empty()) parseAndPublish(line);
122        }
123
124        // 处理不完整的一行
125        if (data.back() != '\n')
126            leftover = data.substr(data.find_last_of('\n') + 1);
127        else
128            leftover.clear();
129
130        ros::spinOnce();
131    }
132
133    close(sockfd);
134}

168void tcpDataLoop(ros::Publisher& raw_pub, 
169                ros::Publisher& roll_pub,
170                ros::Publisher& pitch_pub,
171                ros::Publisher& heading_pub) {
172    char buffer[4096];
173    std::string remaining_data;
174
175    while (ros::ok()) {
176        ssize_t valread = read(g_sockfd, buffer, sizeof(buffer) - 1);
177        if (valread <= 0) {
178            ROS_WARN_STREAM("Read error: " << (valread == 0 ? "EOF" : strerror(errno)));
179            break;
180        }
181
182        buffer[valread] = '\0';
183        std::string raw_data = remaining_data + buffer;
184        size_t last_newline = raw_data.find_last_of('\n');
185
186        if (last_newline != std::string::npos) {
187            // 发布原始数据
188            std_msgs::String raw_msg;
189            raw_msg.data = "[RAW] " + raw_data.substr(0, 100);  // 截断前100字符
190            raw_pub.publish(raw_msg);
191
192            // 解析RPH数据
193            parseAndPublishData(raw_data.substr(0, last_newline), roll_pub, pitch_pub, heading_pub);
194            remaining_data = raw_data.substr(last_newline + 1);
195        } else {
196            remaining_data = raw_data;
197            ROS_DEBUG("Buffering incomplete packet");
198        }
199        memset(buffer, 0, sizeof(buffer));
200    }

203void parseAndPublishData(const std::string& data, 
204                        ros::Publisher& roll_pub,
205                        ros::Publisher& pitch_pub,
206                        ros::Publisher& heading_pub) {
207    std::istringstream iss(data);
208    std::string line;
209    float roll = 0, pitch = 0, heading = 0;
210    bool has_rph = false;
211
212    while (std::getline(iss, line)) {
213        if (line.empty()) continue;
214        ROS_DEBUG_STREAM("Processing: " << line);
215
216        try {
217            if (line.find("R=") == 0) {
218                roll = std::stof(line.substr(2));
219                has_rph = true;
220            } 
221            else if (line.find("P=") == 0) {
222                pitch = std::stof(line.substr(2));
223                has_rph = true;
224            } 
225            else if (line.find("H=") == 0) {
226                heading = std::stof(line.substr(2));
227                if (has_rph) {
228                    std_msgs::Float32 roll_msg, pitch_msg, heading_msg;
229                    roll_msg.data = roll;
230                    pitch_msg.data = pitch;
231                    heading_msg.data = heading;
232                    
233                    roll_pub.publish(roll_msg);
234                    pitch_pub.publish(pitch_msg);
235                    heading_pub.publish(heading_msg);
236                    
237                    ROS_DEBUG_STREAM("Published RPH: R=" << roll 
238                                    << ", P=" << pitch
239                                    << ", H=" << heading);
240                    has_rph = false;
241                }
242            }
243        } catch (const std::exception& e) {
244            ROS_ERROR_STREAM("Parse error: " << e.what() << " in line: " << line);
245        }
246    }
247}
